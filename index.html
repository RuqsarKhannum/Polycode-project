var lastEditedCode = ''; function runMain() { if (running || !editor) return; showTab('io'); var lang = currentLang, code = editor.getValue(); // HTML Web Preview Rendering via iframe if (lang === 'html') { var out = $('output'); out.innerHTML = ''; var doc = document.getElementById('htmlPreview').contentWindow.document; doc.open(); doc.write(code); doc.close(); $('statusLine').textContent = ''; $('statusLine').appendChild(el('span', 'badge ok', 'Rendered')); return; } // Auto-clear STDIN if you changed the code block since the last run if (lastEditedCode && lastEditedCode !== code) { $('stdin').value = ''; } lastEditedCode = code; var stdin = $('stdin').value; // Universal check across any language to see if input functions are used var lowerCode = code.toLowerCase(); var needsInput = lowerCode.includes('input') || lowerCode.includes('readline') || lowerCode.includes('cin') || lowerCode.includes('scanf') || lowerCode.includes('gets') || lowerCode.includes('read') || lowerCode.includes('scanner') || lowerCode.includes('sys.stdin') || lowerCode.includes('getline'); // If the program needs input and the STDIN box is blank, prompt generically if (needsInput && !stdin) { var userInput = window.prompt("Enter input for your program:", ""); if (userInput !== null) { stdin = userInput; $('stdin').value = userInput; // Populates the STDIN panel cleanly } } running = true; $('runBtn').disabled = true; $('statusLine').textContent = ''; $('statusLine').appendChild(el('span', 'badge run', 'Running...')); $('output').textContent = 'Running code on the server...'; $('explainBtn').style.display = 'none'; judge(lang, code, stdin).then(function (res) { lastRun = { lang: lang, code: code, stdin: stdin, res: res }; renderOutput(res); }).catch(function (err) { $('statusLine').textContent = ''; $('output').textContent = friendlyError(err); }).then(function () { running = false; $('runBtn').disabled = false; }); }
